```python daq_counter_output_ni.py
"""Example: Output finite pulse trains from two NI-DAQmx counters at the same time."""

from instro.daq import InstroDAQ
from instro.daq.types import DAQVendor, FrequencyPulseConfig
from instro.lib.publishers import NominalCorePublisher

# Configuration: Choose your vendor. Counter output is supported on NI.
VENDOR = DAQVendor.NI

# Vendor-specific configuration. Each vendor driver lives in its own package and
# owns its transport at construction time.
match VENDOR:
    case DAQVendor.NI:
        from instro.daq.drivers.ni import NIDAQDriver

        # On a 9401 each counter has its own CTR OUT pin: ctr0 on PFI3, ctr1 on PFI7. Any other PFI
        # needs a route, and a second route cannot get the module-wide lock it has to take.
        TERMINAL_0 = "/cDAQ3Mod1/PFI3"
        COUNTER_0 = "cDAQ3/ctr0"
        TERMINAL_1 = "/cDAQ3Mod1/PFI7"
        COUNTER_1 = "cDAQ3/ctr1"
        driver = NIDAQDriver(device_id="cDAQ3")

# Nominal Core dataset to send data to as the instrument is operated.
DATASET_RID = "<dataset_rid>"  # Replace with your dataset RID.

### Main code

daq = InstroDAQ(name="myDAQ", driver=driver)
# daq.add_publisher(NominalCorePublisher(dataset_rid=DATASET_RID))

with daq:
    # Both trains last two seconds: 200 pulses at 100 Hz, and 500 at 250 Hz.
    daq.configure_finite_counter_output(
        physical_channel=TERMINAL_0,
        counter_source=COUNTER_0,
        pulse_config=FrequencyPulseConfig(frequency=100.0, duty_cycle=0.5),
        n_pulses=200,
        alias="clock_0",
    )
    daq.configure_finite_counter_output(
        physical_channel=TERMINAL_1,
        counter_source=COUNTER_1,
        pulse_config=FrequencyPulseConfig(frequency=250.0, duty_cycle=0.25),
        n_pulses=500,
        alias="clock_1",
    )

    # Each counter owns its own task, so the two trains run independently.
    daq.start_counter_output("clock_0")
    daq.start_counter_output("clock_1")

    # Each wait blocks for the train's own duration, plus a second of slack.
    daq.wait_for_counter_output("clock_0")
    daq.wait_for_counter_output("clock_1")

    daq.stop_counter_output("clock_0")
    daq.stop_counter_output("clock_1")
```
